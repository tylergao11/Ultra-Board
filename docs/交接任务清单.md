## T3 清理 data/research 中未被引用的旧草稿
规则：只删除**既未登记为证据、也未被研究库文本引用**的文件。用下面脚本先预览，再加 `--delete` 执行。

```python
# 保存为 data/store/tmp/clean_research.py 运行；结束后删除本脚本
import sqlite3, os, re, sys
root = r"c:/Ai/Ultra-Board"
c = sqlite3.connect(f"file:{root}/data/store/research.sqlite?mode=ro", uri=True)
registered = {r[0] for r in c.execute("select path from evidence_files where path like 'data/research/%'")}
text = []
for q in ["select raw_json from static_research", "select raw_json from dynamic_research",
          "select source_json from research_checks", "select raw_json from research_checks",
          "select sources_json from research_theme_inputs", "select attribute_overrides_json from research_theme_inputs",
          "select raw_json from trade_outcomes"]:
    text += [r[0] or "" for r in c.execute(q)]
refs = set(re.findall(r"data/research/[^\"\\\]\[,]+", "\n".join(text)))
victims = []
for dp, _, files in os.walk(f"{root}/data/research"):
    for f in files:
        rel = os.path.relpath(os.path.join(dp, f), root).replace("\\", "/")
        if rel not in registered and rel not in refs:
            victims.append(rel)
print(len(victims)); [print(v) for v in victims]
if "--delete" in sys.argv:
    for v in victims: os.remove(os.path.join(root, v))
    for dp, dn, fn in sorted(os.walk(f"{root}/data/research"), reverse=True):
        if not os.listdir(dp) and dp != f"{root}/data/research": os.rmdir(dp)
```
- 注意：`tools/research/batch.py` 会读取 `data/research/announcement_checks_2025/`，该目录文件均已登记，脚本不会删除；若预览列表中出现该目录，停止并报告。
